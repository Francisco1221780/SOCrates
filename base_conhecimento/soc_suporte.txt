% ATENCAO: algumas condicoes repetem, pela negativa ou por outras palavras, conhecimento que tambem esta nas regras.
%   sem_condicao_em_falta/3 e o contrario da R18; todas_ao_alcance_nao_afetadas/3 repete o alcance que a R4 calcula;
%   fora_do_destino/2 repete o porto da R4; origem_evidencia/5 usa os vetores da R4 e as instalacoes vulneraveis da R1.
%   Quem mudar uma dessas regras tem de mudar a condicao aqui, e testar os dois lados.
% Estao aqui, e nao nas regras, porque tem de ser calculadas sobre os factos que existem no momento, sem depender da
% ordem em que o motor tira as conclusoes (uma negacao sobre um facto que ainda vai ser deduzido daria o resultado errado).

% Os relogios das fontes (IDS, sensores, logins) nunca estao exatamente certos uns com os outros: ao comparar dois
% tempos aceita-se o desvio registado nos factos em janela(tolerancia_relogios, Segundos). Sem esse facto, a tolerancia e 0.
tolerancia_relogios(Tol) :-
	(   facto(_, janela(tolerancia_relogios, T)) -> Tol = T ; Tol = 0 ).

% depois(Ev2, Ev1, Janela): Ev2 aconteceu depois de Ev1 e dentro da janela (com a tolerancia dos relogios nos dois extremos).
depois(Ev2, Ev1, Janela) :-
	facto(_, tempo(Ev2, T2)),
	facto(_, tempo(Ev1, T1)),
	tolerancia_relogios(Tol),
	T2 >= T1 - Tol,
	T2 - T1 =< Janela + Tol.

% anterior(Ev1, Ev2): Ev1 aconteceu estritamente antes de Ev2 (por isso sao eventos diferentes).
anterior(Ev1, Ev2) :-
	facto(_, tempo(Ev1, T1)),
	facto(_, tempo(Ev2, T2)),
	T1 < T2.

% tardia(Ev2, Ev1, JanelaCurta, JanelaLonga): Ev2 aconteceu depois de Ev1, ja fora da janela curta mas dentro da longa.
tardia(Ev2, Ev1, JanelaCurta, JanelaLonga) :-
	facto(_, tempo(Ev2, T2)),
	facto(_, tempo(Ev1, T1)),
	tolerancia_relogios(Tol),
	D is T2 - T1,
	D > JanelaCurta + Tol,
	D =< JanelaLonga + Tol.

% entre(Outro, Ev1, Ev2): Outro e um evento diferente de Ev1 que aconteceu depois de Ev1 e ate Ev2.
entre(Outro, Ev1, Ev2) :-
	Outro \== Ev1,
	facto(_, tempo(Ev1, T1)),
	facto(_, tempo(Ev2, T2)),
	facto(_, tempo(Outro, T)),
	T > T1, T =< T2.

% sem_ataque_intermedio(EvAtaque, EvHost, Ativo): entre o ataque e a evidencia nao houve outro alerta do IDS contra o
% mesmo ativo. Se houve, a evidencia e mais provavelmente desse outro ataque, e nao conta como tardia do primeiro.
% Se o alerta intermedio so chegar depois, a R69 faz o mesmo raciocinio nessa altura.
sem_ataque_intermedio(EvAtaque, EvHost, Ativo) :-
	facto(_, tempo(EvAtaque, _)),
	facto(_, tempo(EvHost, _)),
	\+ ( facto(_, alerta_ids(Outro, Ativo, _)),
	     entre(Outro, EvAtaque, EvHost) ).

versao_normal(V, v(Ma, Mi, Pa)) :-
	nonvar(V),
	(   V = v(Ma, Mi, Pa) -> true
	;   V = v(Ma, Mi) -> Pa = 0
	),
	versao_campo(Ma), versao_campo(Mi), versao_campo(Pa).

versao_campo(X) :- integer(X), X >= 0.

versao_cmp(A, Op, B) :-
	versao_normal(A, NA),
	versao_normal(B, NB),
	versao_op(Op, NA, NB).

versao_op(<, A, B) :- A @< B.
versao_op(=<, A, B) :- A @=< B.
versao_op(>, A, B) :- A @> B.
versao_op(>=, A, B) :- A @>= B.
versao_op(==, A, B) :- A == B.

versao_invalida(V) :- \+ versao_normal(V, _).
versao_valida(V) :- versao_normal(V, _).

% fora_dos_intervalos(V, Vuln): V e uma versao valida, o catalogo tem pelo menos um intervalo_afetado da vulnerabilidade
% e V nao esta em nenhum deles. Sem intervalos no catalogo nao se conclui nada (R9).
fora_dos_intervalos(V, Vuln) :-
	versao_normal(V, _),
	facto(_, intervalo_afetado(Vuln, _, _)), !,
	\+ ( facto(_, intervalo_afetado(Vuln, Min, Max)),
	     versao_cmp(V, >=, Min),
	     versao_cmp(V, =<, Max) ).

% porto_compativel(Ev, Porto, Protocolo): o servico nesse porto e protocolo pode ter recebido o ataque Ev.
% Se o alerta trouxer destino(Ev, Porto, Protocolo), tem de ser esse; se nao trouxer, qualquer porto serve.
porto_compativel(Ev, Porto, Proto) :-
	(   facto(_, destino(Ev, P0, Pr0))
	->  Porto == P0, Proto == Pr0
	;   true
	).

% sem_servico_acessivel(Ev, Ativo, Programa, Zona): nenhum servico do programa no ativo escuta num porto e protocolo
% abertos para a zona (e, se o alerta Ev trouxer o porto de destino, nesse porto).
sem_servico_acessivel(Ev, Ativo, Nome, Zona) :-
	\+ ( facto(_, servico(_, Ativo, Nome, Porto, Proto)),
	     porto_compativel(Ev, Porto, Proto),
	     facto(_, exposto(Ativo, Porto, Proto, Zona)) ).

% todas_ao_alcance_nao_afetadas(Ev, Ativo, Vuln): todas as instalacoes do programa afetado que o ataque Ev alcanca
% (servico num porto aberto para a zona de origem e, se o alerta o disser, no porto de destino) estao nao afetadas.
% Uma instalacao ao alcance vulneravel, ou de que nao se sabe a versao, chega para a condicao falhar.
todas_ao_alcance_nao_afetadas(Ev, Ativo, Vuln) :-
	facto(_, origem_zona(Ev, Zona)),
	facto(_, vuln_afeta(Vuln, Nome)),
	forall(( facto(_, servico(Inst, Ativo, Nome, Porto, Proto)),
	         porto_compativel(Ev, Porto, Proto),
	         facto(_, exposto(Ativo, Porto, Proto, Zona)) ),
	       facto(_, nao_afetado(Inst, Vuln))).

% fora_do_destino(Ev, Inst): o alerta Ev ja diz o porto de destino, e a instalacao nao escuta nesse porto.
fora_do_destino(Ev, Inst) :-
	facto(_, destino(Ev, Porto, Proto)),
	\+ facto(_, servico(Inst, _, _, Porto, Proto)).

% sem_vetor_exploravel(Ev, Ativo, Vuln): nenhum vetor em vigor (nao revisto) do ataque Ev chega a uma instalacao vulneravel.
sem_vetor_exploravel(Ev, Ativo, Vuln) :-
	\+ ( facto(_, vetor_aberto(Ev, Ativo, Inst, Vuln)),
	     facto(_, vulneravel(Inst, Vuln)),
	     \+ facto(_, revisto(vetor_aberto(Ev, Ativo, Inst, Vuln), _)) ).

% sem_condicao_em_falta(Instalacao, Vuln, Impacto): nenhuma das condicoes que o impacto exige esta registada como "nao"
% na configuracao da instalacao. E o contrario do que a R18 procura.
sem_condicao_em_falta(Inst, Vuln, Impacto) :-
	\+ ( facto(_, requer_condicao(Vuln, Impacto, Cond)),
	     facto(_, configuracao(Inst, Cond, nao)) ).

% evidencias_negadas(Ativo, Ev, Vuln): o ataque Ev tem pelo menos um impacto possivel no ativo (impacto_possivel, grupo E
% das regras), cada impacto da vulnerabilidade no catalogo tem pelo menos uma evidencia esperada, cada evidencia dos
% impactos possiveis tem uma pergunta, e o analista respondeu "nao" a todas essas perguntas.
% Os impactos excluidos pela configuracao (R18) nao contam, mesmo que antes tenham contado como possiveis (R19).
% Se faltar uma evidencia ou uma pergunta no catalogo, falha (e as regras de validacao do catalogo assinalam a falta).
evidencias_negadas(Ativo, Ev, Vuln) :-
	facto(_, impacto_possivel(Ev, Ativo, Vuln, I0)),
	\+ facto(_, impacto_excluido(Ev, Ativo, Vuln, I0, _)), !,
	forall(facto(_, vuln_impacto(Vuln, Imp)),
	       facto(_, evidencia_impacto(Imp, _, _))),
	forall(( facto(_, impacto_possivel(Ev, Ativo, Vuln, Imp)),
	         \+ facto(_, impacto_excluido(Ev, Ativo, Vuln, Imp, _)),
	         facto(_, evidencia_impacto(Imp, Tipo, _)) ),
	       ( facto(_, pergunta_evidencia(Tipo, P)),
	         facto(_, resposta_analista(Ativo, Ev, P, nao)) )).

% origem_evidencia(EvHost, EvAtaque, Ativo, Vuln, Origem): de onde veio um evento do host (uma evidencia, ou o trafego
% para um C2) em relacao ao ataque EvAtaque a vulnerabilidade Vuln. Origem e uma de:
%   ligada                                    veio da instancia de processo de uma instalacao vulneravel que o ataque alcancou;
%   de_outro_processo(programa(P))            veio de um processo de outro programa (P);
%   de_outro_processo(instancia(I))           veio de outra instancia (I), que o ataque nao alcancou, do mesmo programa;
%   por_confirmar(sem_processo)               o sensor nao disse que processo foi;
%   por_confirmar(processo_desconhecido(N))   o processo N nao esta na tabela processo_componente;
%   por_confirmar(instancia_por_identificar(N))  ha varias instancias de N no ativo e o sensor nao disse qual foi;
%   por_confirmar(instancia_nao_inventariada(N)) N e do programa atacado, mas nao ha nenhuma instancia dele inventariada no ativo.
% A instancia vem do sensor (evento_instancia); se o sensor so disser o nome, conta a unica instancia com esse nome no ativo.
origem_evidencia(EvHost, EvAtaque, Ativo, Vuln, Origem) :-
	once(facto(_, vuln_afeta(Vuln, Prog))),
	classifica_origem(EvHost, EvAtaque, Ativo, Vuln, Prog, O),
	Origem = O.

classifica_origem(EvHost, EvAtaque, Ativo, Vuln, _, O) :-
	once(facto(_, evento_instancia(EvHost, I))), !,
	instancia_ligada(EvAtaque, Ativo, Vuln, I, O).
classifica_origem(EvHost, EvAtaque, Ativo, Vuln, Prog, O) :-
	once(facto(_, evento_processo(EvHost, Nome))), !,
	(   \+ facto(_, processo_componente(_, Nome))
	->  O = por_confirmar(processo_desconhecido(Nome))
	;   \+ facto(_, processo_componente(Prog, Nome))
	->  once(facto(_, processo_componente(Outro, Nome))),
	    O = de_outro_processo(programa(Outro))
	;   instancias_com_nome(Ativo, Nome, Is),
	    (   Is = [I] -> instancia_ligada(EvAtaque, Ativo, Vuln, I, O)
	    ;   Is = []  -> O = por_confirmar(instancia_nao_inventariada(Nome))
	    ;   O = por_confirmar(instancia_por_identificar(Nome))
	    )
	).
classifica_origem(_, _, _, _, _, por_confirmar(sem_processo)).

instancia_ligada(EvAtaque, Ativo, Vuln, I, O) :-
	(   instancia_atacada(EvAtaque, Ativo, Vuln, I) -> O = ligada ; O = de_outro_processo(instancia(I)) ).

% instancia_atacada(EvAtaque, Ativo, Vuln, I): I e a instancia de processo de uma instalacao vulneravel a que chega um
% vetor em vigor (nao revisto) do ataque.
instancia_atacada(EvAtaque, Ativo, Vuln, I) :-
	facto(_, vetor_aberto(EvAtaque, Ativo, Inst, Vuln)),
	\+ facto(_, revisto(vetor_aberto(EvAtaque, Ativo, Inst, Vuln), _)),
	facto(_, vulneravel(Inst, Vuln)),
	facto(_, instancia_processo(Inst, I)), !.

% instancias_com_nome(Ativo, Nome, Instancias): as instancias de processo do ativo cujo programa corre com esse nome.
instancias_com_nome(Ativo, Nome, Is) :-
	findall(I, ( facto(_, instancia_processo(Inst, I)),
	             facto(_, componente(Inst, Ativo, P)),
	             facto(_, processo_componente(P, Nome)) ), Is0),
	sort(Is0, Is).

% em_aberto(Incidente): o incidente nao foi superado nem resolvido.
em_aberto(Inc) :-
	\+ facto(_, superado(Inc, _)),
	\+ facto(_, resolvido(Inc, _)).

% todos_motivos_revistos(Incidente): o incidente tem pelo menos um motivo, e todos os seus motivos passaram a historicos.
todos_motivos_revistos(Inc) :-
	facto(_, motivo(Inc, _)), !,
	forall(facto(_, motivo(Inc, M)), facto(_, revisto(motivo(Inc, M), _))).

% novo_facto(F): acrescenta o facto F com o proximo numero livre. E o que os ficheiros da pasta cenarios/ usam para
% injetar a telemetria e as respostas do analista: os factos deduzidos pelo motor usam o mesmo contador, por isso os
% numeros nunca se repetem. Se o facto ja existir nao faz nada (carregar o mesmo cenario duas vezes nao o repete).
novo_facto(F) :-
	facto(_, F), !.
novo_facto(F) :-
	retract(ultimo_facto(N0)),
	N is N0 + 1,
	assertz(ultimo_facto(N)),
	assertz(facto(N, F)).
