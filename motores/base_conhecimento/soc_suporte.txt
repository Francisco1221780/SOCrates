% sem_condicao_em_falta(Instalacao, Vuln, Impacto): nenhuma das condicoes que o impacto exige esta registada como "nao"
% na configuracao da instalacao. E o contrario do que a R18 procura. Calcula-se sobre os factos do catalogo e do
% inventario para nao depender da ordem em que o motor tira as conclusoes.
sem_condicao_em_falta(Inst, Vuln, Impacto) :-
	\+ ( facto(_, requer_condicao(Vuln, Impacto, Cond)),
	     facto(_, configuracao(Inst, Cond, nao)) ).

% ATENCAO: tres destas condicoes repetem, pela negativa, conhecimento que tambem esta escrito nas regras.
%   sem_ligacao_ao_ataque/2 e o contrario da ligacao exigida pela R40; origem_desconhecida/1 e o contrario das origens
%   reconhecidas pela R20 e pela R21; sem_condicao_em_falta/3 e o contrario da R18.
%   Quem mudar uma dessas regras tem de mudar a condicao aqui, e testar os dois lados.

% Os relogios das fontes (IDS, sensores, logins) nunca estao exatamente certos uns com os outros: ao comparar dois
% tempos aceita-se o desvio registado nos factos em janela(tolerancia_relogios, Segundos). Sem esse facto, a tolerancia e 0.
tolerancia_relogios(Tol) :-
	(   facto(_, janela(tolerancia_relogios, T)) -> Tol = T ; Tol = 0 ).

% depois(Ev2, Ev1, Janela): Ev2 aconteceu depois de Ev1 e dentro da janela (com a tolerancia dos relogios nos dois extremos).
depois(Ev2, Ev1, Janela) :-
	facto(_, tempo(Ev2, T2)),
	facto(_, tempo(Ev1, T1)),
	tolerancia_relogios(Tol),
	T2 >= T1 - Tol,
	T2 - T1 =< Janela + Tol.

% anterior(Ev1, Ev2): Ev1 aconteceu estritamente antes de Ev2 (por isso sao eventos diferentes).
anterior(Ev1, Ev2) :-
	facto(_, tempo(Ev1, T1)),
	facto(_, tempo(Ev2, T2)),
	T1 < T2.

% tardia(Ev2, Ev1, JanelaCurta, JanelaLonga): Ev2 aconteceu depois de Ev1, ja fora da janela curta mas dentro da longa.
tardia(Ev2, Ev1, JanelaCurta, JanelaLonga) :-
	facto(_, tempo(Ev2, T2)),
	facto(_, tempo(Ev1, T1)),
	tolerancia_relogios(Tol),
	D is T2 - T1,
	D > JanelaCurta + Tol,
	D =< JanelaLonga + Tol.

% sem_ataque_intermedio(EvAtaque, EvHost, Ativo): entre o ataque e a evidencia nao houve outro alerta do IDS contra o
% mesmo ativo. Se houve, a evidencia e mais provavelmente desse outro ataque, e nao conta como tardia do primeiro.
sem_ataque_intermedio(EvAtaque, EvHost, Ativo) :-
	facto(_, tempo(EvAtaque, T1)),
	facto(_, tempo(EvHost, T2)),
	\+ ( facto(_, alerta_ids(Outro, Ativo, _)),
	     Outro \== EvAtaque,
	     facto(_, tempo(Outro, T)),
	     T > T1, T =< T2 ).

versao_normal(V, v(Ma, Mi, Pa)) :-
	nonvar(V),
	(   V = v(Ma, Mi, Pa) -> true
	;   V = v(Ma, Mi) -> Pa = 0
	),
	versao_campo(Ma), versao_campo(Mi), versao_campo(Pa).

versao_campo(X) :- integer(X), X >= 0.

versao_cmp(A, Op, B) :-
	versao_normal(A, NA),
	versao_normal(B, NB),
	versao_op(Op, NA, NB).

versao_op(<, A, B) :- A @< B.
versao_op(=<, A, B) :- A @=< B.
versao_op(>, A, B) :- A @> B.
versao_op(>=, A, B) :- A @>= B.
versao_op(==, A, B) :- A == B.

versao_invalida(V) :- \+ versao_normal(V, _).

% fora_dos_intervalos(V, Vuln): V e uma versao valida, o catalogo tem pelo menos um intervalo_afetado da vulnerabilidade
% e V nao esta em nenhum deles. Sem intervalos no catalogo nao se conclui nada (R9).
fora_dos_intervalos(V, Vuln) :-
	versao_normal(V, _),
	facto(_, intervalo_afetado(Vuln, _, _)), !,
	\+ ( facto(_, intervalo_afetado(Vuln, Min, Max)),
	     versao_cmp(V, >=, Min),
	     versao_cmp(V, =<, Max) ).

% porto_compativel(Ev, Porto, Protocolo): o servico nesse porto e protocolo pode ter recebido o ataque Ev.
% Se o alerta trouxer destino(Ev, Porto, Protocolo), tem de ser esse; se nao trouxer, qualquer porto serve.
porto_compativel(Ev, Porto, Proto) :-
	(   facto(_, destino(Ev, P0, Pr0))
	->  Porto == P0, Proto == Pr0
	;   true
	).

% sem_servico_acessivel(Ev, Ativo, Programa, Zona): nenhum servico do programa no ativo escuta num porto e protocolo
% abertos para a zona (e, se o alerta Ev trouxer o porto de destino, nesse porto).
sem_servico_acessivel(Ev, Ativo, Nome, Zona) :-
	\+ ( facto(_, servico(_, Ativo, Nome, Porto, Proto)),
	     porto_compativel(Ev, Porto, Proto),
	     facto(_, exposto(Ativo, Porto, Proto, Zona)) ).

% evidencias_negadas(Ativo, Ev, Vuln): o ataque Ev tem pelo menos um impacto possivel no ativo (impacto_possivel, grupo E
% das regras), cada impacto da vulnerabilidade no catalogo tem pelo menos uma evidencia esperada, cada evidencia dos
% impactos possiveis tem uma pergunta, e o analista respondeu "nao" a todas essas perguntas.
% Os impactos excluidos pela configuracao (R18) nao contam, mesmo que antes tenham contado como possiveis (R19).
% Se faltar uma evidencia ou uma pergunta no catalogo, falha (e as regras de validacao do catalogo assinalam a falta).
evidencias_negadas(Ativo, Ev, Vuln) :-
	facto(_, impacto_possivel(Ev, Ativo, Vuln, I0)),
	\+ facto(_, impacto_excluido(Ev, Ativo, Vuln, I0, _)), !,
	forall(facto(_, vuln_impacto(Vuln, Imp)),
	       facto(_, evidencia_impacto(Imp, _, _))),
	forall(( facto(_, impacto_possivel(Ev, Ativo, Vuln, Imp)),
	         \+ facto(_, impacto_excluido(Ev, Ativo, Vuln, Imp, _)),
	         facto(_, evidencia_impacto(Imp, Tipo, _)) ),
	       ( facto(_, pergunta_evidencia(Tipo, P)),
	         facto(_, resposta_analista(Ativo, Ev, P, nao)) )).

% sem_ligacao_ao_ataque(EvC2, Vuln): nao se conhece ligacao entre o trafego para o C2 e o ataque a vulnerabilidade Vuln:
% nao ha nenhum evento_processo para o trafego que seja um processo do programa atacado (ou nao ha telemetria de
% processo, ou o processo e de outro programa ou nao esta na tabela). E o contrario da ligacao exigida pela R40.
sem_ligacao_ao_ataque(EvC2, Vuln) :-
	\+ ( facto(_, evento_processo(EvC2, Proc)),
	     facto(_, vuln_afeta(Vuln, Nome)),
	     facto(_, processo_componente(Nome, Proc)) ).

% origem_desconhecida(EvHost): nao se sabe que programa originou o evento do host: nao ha nenhum evento_processo para ele,
% ou ha, mas o processo nao esta em nenhum processo_componente.
origem_desconhecida(EvHost) :-
	(   \+ facto(_, evento_processo(EvHost, _))
	->  true
	;   facto(_, evento_processo(EvHost, Proc)),
	    \+ facto(_, processo_componente(_, Proc))
	->  true
	).

% novo_facto(F): acrescenta o facto F com o proximo numero livre. Usa-se para injetar a telemetria dos cenarios e as
% respostas do analista: os factos deduzidos pelo motor usam o mesmo contador, por isso os numeros nunca se repetem.
novo_facto(F) :-
	retract(ultimo_facto(N0)),
	N is N0 + 1,
	assertz(ultimo_facto(N)),
	assertz(facto(N, F)).
